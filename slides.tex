\begin{frame}[fragile]
\frametitle{Monoid}
\begin{center}
\begin{minipage}{0.6\textwidth}
\begin{haskellcode}
class Monoid m where
  mappend :: m -> m -> m
  mempty  :: m
\end{haskellcode}
\pause
\begin{haskellcode}
instance Monoid String where
  mappend s t = s ++ t
  mempty = ""
\end{haskellcode}
\end{minipage}
\end{center}
\end{frame}


\begin{frame}[fragile]
\frametitle{Monoid laws}
\begin{center}
\begin{minipage}{0.6\textwidth}
Left identity:

\begin{haskellcode}
-- for all x.
mappend mempty x == x
\end{haskellcode}

\hrule

\bigskip

Right identity:

\begin{haskellcode}
-- for all x.
mappend x mempty == x
\end{haskellcode}

\hrule

\bigskip

Associativity:

\begin{haskellcode}
-- for all x y z.
mappend x (mappend y z) == mappend (mappend x y) z
\end{haskellcode}

\end{minipage}
\end{center}
\end{frame}
