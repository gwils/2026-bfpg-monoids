\begin{frame}[fragile]
\frametitle{Monoid}
\begin{center}
\begin{minipage}{0.6\textwidth}
\begin{haskellcode}
class Monoid m where
  mappend :: m -> m -> m
  mempty  :: m
\end{haskellcode}
\pause
\begin{haskellcode}
-- mappend as an infix operator, for convenience
x <> y = mappend x y
\end{haskellcode}
\pause
\begin{haskellcode}
instance Monoid String where
  mappend s t = s ++ t
  mempty = ""
\end{haskellcode}
\end{minipage}
\end{center}
\end{frame}


\begin{frame}[fragile]
\frametitle{Monoid laws}
\begin{center}
\begin{minipage}{0.6\textwidth}
Left identity:

\begin{haskellcode}
-- for all x.
mempty <> x == x
\end{haskellcode}

\pause
\hrule
\bigskip

Right identity:

\begin{haskellcode}
-- for all x.
x <> mempty == x
\end{haskellcode}

\pause
\hrule
\bigskip

Associativity:

\begin{haskellcode}
-- for all x y z.
x <> (y <> z) == (x <> y) <> z
\end{haskellcode}

\end{minipage}
\end{center}
\end{frame}


\begin{frame}[fragile]
\frametitle{Monoid laws specialised to strings}
\begin{center}
\begin{minipage}{0.6\textwidth}
Left identity:

\begin{haskellcode}
-- for all x.
"" ++ x == x
\end{haskellcode}

\hrule
\bigskip

Right identity:

\begin{haskellcode}
-- for all x.
x ++ "" == x
\end{haskellcode}

\hrule
\bigskip

Associativity:

\begin{haskellcode}
-- for all x y z.
x ++ (y ++ z) == (x ++ y) ++ z
\end{haskellcode}

\end{minipage}
\end{center}
\end{frame}

\begin{frame}[fragile]
\frametitle{Trees}
\begin{center}
\begin{minipage}{0.6\textwidth}
\begin{haskellcode}
data Tree a = Leaf a | Branch (Tree a) (Tree a)

\end{haskellcode}
\pause
\begin{haskellcode}
concatTree :: Monoid m => Tree m -> m
concatTree t =
  case t of
    Leaf m -> m
    Branch t u -> concatTree t <> concatTree u
\end{haskellcode}
\end{minipage}
\end{center}
\end{frame}


\begin{frame}[fragile]
\frametitle{Derived functions}
\begin{center}
\begin{minipage}{0.6\textwidth}
\begin{haskellcode}
class Foldable f where
  fold :: Monoid m => f m -> m
\end{haskellcode}
\pause
\begin{haskellcode}
  -- much more exciting
  foldMap :: Monoid m => (a -> m) -> f a -> m
\end{haskellcode}
\pause
\begin{haskellcode}
instance Foldable [] where
  foldMap a2m list = mconcat (map a2m list)
\end{haskellcode}
\pause
\begin{haskellcode}
instance Foldable Tree where
  foldMap a2m tree =
    case tree of
      Leaf a -> a2m a
      Branch t u -> foldMap a2m t <> foldMap a2m u
\end{haskellcode}
\end{minipage}
\end{center}
